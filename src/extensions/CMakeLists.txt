set_property(GLOBAL PROPERTY GMRUNNER_EXTENSION_TARGETS "")
set_property(GLOBAL PROPERTY GMRUNNER_EXTENSION_ENTRIES "")

# ---------------------------------------------------------------------------
# Core extension framework & shared service headers
# ---------------------------------------------------------------------------
add_library(gm_extensions_core STATIC
    core/extension_host.h
    core/extension_services.h
    core/extension_context.h
    core/extension_exports.h
    core/extension.h
    core/extension_registry.h
    core/extension_registry.cpp
    shared/buffer_store.h
)

target_include_directories(gm_extensions_core PUBLIC ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(gm_extensions_core
    PUBLIC
        gm_base
        gm_platform
        gm_loader
        gm_gml
        gm_graphics
        gm_audio
)
gmrunner_apply_warnings(gm_extensions_core)

# ---------------------------------------------------------------------------
# Auto-discover extension modules under src/extensions/modules/<name>/
# ---------------------------------------------------------------------------
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/ExtensionHelpers.cmake)

file(GLOB _gm_ext_cmakelists CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/modules/*/CMakeLists.txt"
)
list(SORT _gm_ext_cmakelists)

foreach(_ext_cmake IN LISTS _gm_ext_cmakelists)
    get_filename_component(_ext_dir "${_ext_cmake}" DIRECTORY)
    add_subdirectory("${_ext_dir}")
endforeach()

# ---------------------------------------------------------------------------
# Generate deterministic compile-time registration glue (safe against static
# library linker dead-stripping across MSVC, GCC, Clang, Emscripten, and NDK)
# ---------------------------------------------------------------------------
get_property(_gm_ext_targets GLOBAL PROPERTY GMRUNNER_EXTENSION_TARGETS)
get_property(_gm_ext_entries GLOBAL PROPERTY GMRUNNER_EXTENSION_ENTRIES)
list(SORT _gm_ext_entries)

set(_ext_decls "")
set(_ext_calls "")
foreach(_entry IN LISTS _gm_ext_entries)
    string(REPLACE ":" ";" _parts "${_entry}")
    list(GET _parts 2 _factory)
    string(APPEND _ext_decls "std::unique_ptr<IExtension> ${_factory}();\n")
    string(APPEND _ext_calls "    register_extension(${_factory}());\n")
endforeach()

set(_generated_cpp "${CMAKE_CURRENT_BINARY_DIR}/generated_extension_registry.cpp")
file(CONFIGURE OUTPUT "${_generated_cpp}" CONTENT
"// Auto-generated by src/extensions/CMakeLists.txt — do not edit manually.
#include \"extensions/core/extension_registry.h\"

#include <memory>

namespace gm {

@_ext_decls@
void ExtensionRegistry::register_compiled_extensions() {
    clear_extensions();
@_ext_calls@    log_registered_summary();
}

} // namespace gm
" @ONLY)

# ---------------------------------------------------------------------------
# Aggregate target consumed by gm_runtime
# ---------------------------------------------------------------------------
add_library(gm_extensions STATIC
    "${_generated_cpp}"
)
target_link_libraries(gm_extensions
    PRIVATE
        ${_gm_ext_targets}
    PUBLIC
        gm_extensions_core
)
gmrunner_apply_warnings(gm_extensions)
